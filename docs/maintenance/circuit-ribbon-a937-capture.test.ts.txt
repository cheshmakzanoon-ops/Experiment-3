import { it } from 'vitest';
import { readFileSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { gzipSync } from 'node:zlib';
import { execFileSync } from 'node:child_process';
import * as T from 'three';
import { ribbonOracleCases, renderRibbonCase } from './fixtures/circuit-ribbon-cases.ts';
const hash = (bytes: string | Uint8Array) => createHash('sha256').update(bytes).digest('hex');
it('captures unmodified a937 Three oracle exactly once', () => {
  const commit = 'a937cacf028ae81e3bd6c6fa3d4634f2295e4886';
  const source = readFileSync('src/rendering/circuit.ts');
  if (!source.equals(execFileSync('git', ['show', `${commit}:src/rendering/circuit.ts`])))
    throw new Error('Baseline source differs');
  const array = (value: ArrayBufferView) => ({
    type: value.constructor.name,
    bytes: Buffer.from(value.buffer, value.byteOffset, value.byteLength).toString('base64'),
  });
  const cases = ribbonOracleCases().map((entry) => {
    const built = renderRibbonCase(entry);
    const result = {
      name: entry.name,
      circuit: entry.track.circuit.id,
      start: entry.options.start,
      end: entry.options.end,
      first: built.meshes.indexOf(built.first),
      meshes: built.meshes.map((mesh) => ({
        attributes: Object.fromEntries(
          Object.entries(mesh.geometry.attributes).map(([name, attr]) => [
            name,
            { itemSize: attr.itemSize, normalized: attr.normalized, ...array(attr.array) },
          ]),
        ),
        index: array(mesh.geometry.index!.array),
        drawRange: mesh.geometry.drawRange,
        groups: mesh.geometry.groups,
        castShadow: mesh.castShadow,
        receiveShadow: mesh.receiveShadow,
        strip: mesh.userData.strip
          ? {
              width: mesh.userData.strip.width,
              centreline: mesh.userData.strip.centreline.map((v: T.Vector3) => v.toArray()),
            }
          : null,
      })),
    };
    built.meshes.forEach((mesh) => mesh.geometry.dispose());
    built.material.dispose();
    return result;
  });
  const evidence = {
    sourceCommit: commit,
    sourceFileSHA256: hash(source),
    caseFixtureSHA256: hash(readFileSync('tests/fixtures/circuit-ribbon-cases.ts')),
    threeRevision: T.REVISION,
    nodeVersion: process.version,
    cases,
  };
  const output = gzipSync(JSON.stringify(evidence), { level: 9 });
  writeFileSync('tests/fixtures/circuit-ribbon-a937.json.gz', output);
  console.log(
    JSON.stringify({
      sourceCommit: commit,
      sourceSHA256: evidence.sourceFileSHA256,
      fixtureSHA256: evidence.caseFixtureSHA256,
      gzipSHA256: hash(output),
      cases: cases.length,
      meshes: cases.reduce((n, c) => n + c.meshes.length, 0),
      bytes: output.length,
    }),
  );
});
